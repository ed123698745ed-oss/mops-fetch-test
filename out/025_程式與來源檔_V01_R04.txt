# 逐日抓公開資訊觀測站「每日重大訊息」(t05st02)，保存原始 HTML(gzip) 並解析成 CSV
# 用法：python fetch_year.py 2023            （整年）
#       python fetch_year.py 20230505 20230623 （指定起訖日，含兩端）
import sys, os, re, gzip, csv, time, datetime, html as H, urllib.request, urllib.parse
a = sys.argv[1]
dates = None
if a.startswith("dates:"):       # 指定個別日期：dates:20230726,20230729
    dates = [datetime.datetime.strptime(x, "%Y%m%d").date() for x in a[6:].split(",")]
    start, end = dates[0], dates[-1]; tag = "dates_" + a[6:].split(",")[0] + "_" + str(len(dates))
elif len(a) == 4:
    start, end = datetime.date(int(a),1,1), datetime.date(int(a),12,31); tag = a
else:
    P = lambda x: datetime.datetime.strptime(x, "%Y%m%d").date()
    start, end = P(a), P(sys.argv[2]); tag = f"{a}_{sys.argv[2]}"
year = start.year
URL = "https://mopsov.twse.com.tw/mops/web/ajax_t05st02"
UA = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/129.0 Safari/537.36"
rawdir = f"raw/{year}"; os.makedirs(rawdir, exist_ok=True); os.makedirs("data", exist_ok=True)
log = open(f"data/fetch_log_{tag}.txt", "a", encoding="utf-8")
def L(s): print(s); log.write(s + "\n"); log.flush()

def fetch(d):
    body = urllib.parse.urlencode({"encodeURIComponent":1,"step":1,"step00":0,"firstin":1,"off":1,
        "TYPEK":"all","year":d.year-1911,"month":f"{d.month:02d}","day":f"{d.day:02d}"}).encode()
    req = urllib.request.Request(URL, data=body, headers={"User-Agent":UA,"Referer":"https://mopsov.twse.com.tw/mops/web/t05st02"})
    return urllib.request.urlopen(req, timeout=90).read().decode("utf-8","replace")

form_re = re.compile(r"<form action='/mops/web/ajax_t05st02'.*?</form>", re.S)
inp_re = re.compile(r"<input type='hidden' name='h(\d+)' value='(.*?)'>", re.S)
typ_re = re.compile(r'TYPEK\.value="(\w+)"')
def parse(t, qdate):
    rows = []
    for f in form_re.findall(t):
        v = {"h1"+k[-1]: H.unescape(x) for k, x in inp_re.findall(f)}   # 欄位名 h<序號><0-8>，取最後一位
        m = typ_re.search(f)
        if "h11" not in v: continue
        rows.append([qdate, m.group(1) if m else "", v.get("h11",""), v.get("h10",""), v.get("h12",""), v.get("h13",""),
                     v.get("h14",""), v.get("h15",""), v.get("h16",""), v.get("h17",""), v.get("h18","")])
    return rows

out = open(f"data/announcements_{tag}.csv", "w", encoding="utf-8-sig", newline="")
w = csv.writer(out); w.writerow(["查詢日","市場別","公司代號","公司名稱","發言日期","發言時間","主旨","條款代號h15","款項h16","事實發生日","說明"])
d = start; n_total = 0
todo = dates if dates else None
while d <= end and d <= datetime.date.today():
    if todo is not None and d not in todo:
        d += datetime.timedelta(days=1); continue
    rawdir = f"raw/{d.year}"; os.makedirs(rawdir, exist_ok=True)
    f = f"{rawdir}/t05st02_{d:%Y%m%d}.html.gz"
    for attempt in range(3):
        try:
            t = fetch(d)
            if "SECURITY" in t[:3000]:
                L(f"{d} 被擋(安全性錯誤頁)"); raise SystemExit(2)
            gzip.open(f, "wt", encoding="utf-8").write(t)
            rows = parse(t, d.strftime("%Y%m%d")); w.writerows(rows); out.flush(); n_total += len(rows)
            L(f"{d} 原始{len(t)}字元 公告{len(rows)}筆"); break
        except SystemExit: raise
        except Exception as e:
            L(f"{d} 第{attempt+1}次錯誤 {e}"); time.sleep(10)
    time.sleep(3)
    d += datetime.timedelta(days=1)
out.close(); L(f"完成 {tag} 共解析 {n_total} 筆(未去重)")

# 合併所有抓取分段的公告 CSV，依年份輸出去重後的公告總表，並列出每年缺漏日期
# 用法：python merge_all.py
import csv, gzip, glob, os, datetime
from collections import defaultdict

os.makedirs("merged", exist_ok=True)
hdr, by_year, seen = None, defaultdict(list), set()
for f in sorted(glob.glob("data/announcements_*.csv.gz")):
    for r in csv.reader(gzip.open(f, "rt", encoding="utf-8-sig")):
        if r[0] == "查詢日":
            hdr = r; continue
        k = (r[2], r[4], r[5], r[6])          # 公司代號、發言日期、發言時間、主旨
        if k in seen:
            continue
        seen.add(k); by_year[r[4][:4]].append(r)

for y, rows in sorted(by_year.items()):
    rows.sort(key=lambda r: (r[4], r[5], r[2]))
    with gzip.open(f"merged/MOPS每日重大訊息_公告總表_{y}.csv.gz", "wt", encoding="utf-8-sig", newline="") as g:
        w = csv.writer(g); w.writerow(hdr); w.writerows(rows)
    print(y, "去重後", len(rows), "上市櫃", sum(r[1] in ("sii", "otc") for r in rows))

# 缺漏日期：以 raw 原始檔是否存在判斷（查詢日）
for ydir in sorted(glob.glob("raw/*")):
    y = int(os.path.basename(ydir))
    have = {os.path.basename(p)[8:16] for p in glob.glob(f"{ydir}/*.html.gz")}
    d, miss = datetime.date(y, 1, 1), []
    while d.year == y and d <= datetime.date.today():
        if d.strftime("%Y%m%d") not in have:
            miss.append(d.strftime("%m%d"))
        d += datetime.timedelta(days=1)
    print(y, "已抓", len(have), "天，缺", len(miss), "天", (miss if len(miss) <= 40 else miss[:5] + ["…"] + miss[-5:]))

import re

def norm(s):
    """主旨常被換行切斷字詞（如「股\n權」），一律去除空白與換行再比對"""
    import unicodedata
    return re.sub(r"\s+", "", unicodedata.normalize("NFKC", s or ""))   # R1.1：NFKC 統一異體字(如康熙部首「⼟」→「土」)

def is_candidate(subj, body):
    subj, body = norm(subj), norm(body)
    hit_subj = bool(TERM.search(subj) and OBJ.search(subj))
    hit_body = bool(TERM.search(body) and OBJ.search(body)) and bool(TERM.search(subj))

def role_of(subj, body):
    t = subj + norm(body)[:300]
    if re.search(r"被授權|取得.*授權", t): return "被授權方"
    if re.search(r"授權予|授權他人|授權方", t): return "授權方"
    if re.search(r"採購|發包|訂購|委建|委託.*(營造|廠商)", t): return "採購方"
    if re.search(r"經銷|代理|銷售|供貨|供應|代工", t): return "供貨/服務提供者"
    return "不明"

def classify(subj, body, h15="", h16=""):
    """回傳 dict：判讀(符合/符合-工程/符合-租賃出租方/排除/待人工)、類別、合約類型、終止時點、公司角色、規則備註"""
    subj, body = norm(subj), norm(body)
    out = {"判讀": "", "類別": "", "合約類型": "", "終止時點": "", "公司角色": "", "規則備註": ""}
    eng = bool(ENG.search(ENG_NAME.sub("", subj)))
    # 1. 租賃：出租方且為重要客戶 → 納入並標記；其餘租約排除
    if LEASE.search(subj):
        if CUSTOMER.search(subj):
            out.update(判讀="符合-租賃出租方", 類別="租賃出租方/重要客戶", 合約類型="商業合約",
                       規則備註="Ed 2026-10-19：納入，單獨標記")
            return _timing(out, subj, body)
        out.update(判讀="排除", 類別="租約/使用權資產"); return out
    # 2. 排除類別（工程承攬合約內含「合建」「融資」「投資興建」字樣者不因此排除）
    for name, rx in EXCLUDE_RULES:
        if rx.search(subj):
            if eng and name in ("股權/合資/併購/投資", "融資/保證"):
                continue
            out.update(判讀="排除", 類別=name); return out
    if CODEV.search(subj) or (REALESTATE.search(subj) and not eng and not LICENSE.search(subj)):
        out.update(判讀="排除", 類別="不動產/土地合建"); return out
    # 3. 需人工的情形
    if GENERIC.search(subj):
        out.update(判讀="待人工", 類別="主旨未載合約對象", 規則備註="需讀說明文字")
        return _timing(out, subj, body)
    if ASSET.search(subj) and not LICENSE.search(subj):
        out.update(判讀="排除", 類別="資產買賣(處分/取得)", 規則備註="Elon 補的邊界：資產買賣合約不屬商業合約/授權，待 Ed 確認")
        return out
    if UNCERTAIN.search(subj) and not DEFINITE.search(body[:800]):
        out.update(判讀="待人工", 類別="非確定/擬終止", 規則備註="卡 6：僅稱可能終止/協商者不納入；擬/預計需人工看是否為確定決議")
        return _timing(out, subj, body)
    # 4. 納入
    if PARTNER_STOP.search(subj):
        if re.search(r"租賃", body[:800]):
            out.update(判讀="符合-租賃出租方", 類別="租賃出租方/重要客戶", 合約類型="商業合約", 規則備註="MOPS「重要客戶/供應商終止往來」款項；說明為租賃合約到期不續約；Ed 2026-10-19：出租方納入並標記")
        else:
            out.update(判讀="符合", 類別="商業合約", 合約類型="商業合約", 規則備註="MOPS「重要客戶/供應商終止往來」款項")
        return _timing(out, subj, body)
    if TERM.search(subj) and OBJ.search(subj):
        if TRANSFER_AGENT.search(subj):
            out.update(判讀="排除", 類別="股務代理業務(服務契約)", 規則備註="Ed 2026-10-20：股務代理為公司付費的行政服務契約，不影響營收，不納入")

#  3. 預定出場日＝E 起第 H 個市場交易日；當日個股無有效收盤 → 順延至第一個有有效收盤的市場交易日，記實際持有日數。
#     若在順延期間已終止上市櫃 → 以終止前最後一個有效收盤（最終交易）結算並標記；仍無價 → 列「未解決」，不刪除、不填零。
#     預定出場日晚於資料終點 → 列「未完成（尚未到出場日）」。
#  4. 持有期間（不含進場日）出現無法以除權息解釋的跳價（超過當時漲跌幅限制 1.5 個百分點以上）→ 列「未解決：疑似公司行動未核對」，不納入統計。
#  5. 可成交性：進場日或出場日為「一字價且達漲跌停」者，標記為不可證明成交；理論價格樣本全列，另列可成交樣本。
#  6. 另列：七窗口共同完整樣本、中位數與極端值、年度與個股分布、同股持倉期間不新增部位的敏感度。

def stock(sid):
    if sid not in CACHE:
        px = {x["date"]: x for x in load(sid)}
        div = load(sid, "TaiwanStockDividend")
        ev = {}
        for x in load(sid, "TaiwanStockDividendResult"):
            d, typ, val = x["date"], x["stock_or_cache_dividend"], x["stock_and_cache_dividend"]
            if typ in ("息", "除息"):
                ev[d] = dict(m=1.0, c=val, note=f"現金股利{val}元（FinMind 除息結果）")
                continue
            rows = [y for y in div if d in (y["StockExDividendTradingDate"], y["CashExDividendTradingDate"])]
            if len(rows) != 1: ev[d] = dict(bad=f"{typ}：無法對應唯一股利政策（{len(rows)}筆），無法拆分現金與配股", val=val, bef=x["before_price"], aft=x["after_price"]); continue
            y = rows[0]; st = (y["StockEarningsDistribution"] + y["StockStatutorySurplus"]) / 10.0; ca = y["CashEarningsDistribution"] + y["CashStatutorySurplus"]
            if y["CashIncreaseSubscriptionRate"] > 0:
                ev[d] = dict(bad=f"{typ}：現金增資認購權（每千股{y['CashIncreaseSubscriptionRate']:.4g}股、認購價{y['CashIncreaseSubscriptionpRrice']}），權利處置無資料", val=val, bef=x["before_price"], aft=x["after_price"], st=st, ca=ca); continue
            if abs((x["before_price"] - ca) / (1 + st) - x["after_price"]) > 0.05:
                ev[d] = dict(bad=f"{typ}：股利政策與參考價不符，無法核對", val=val, bef=x["before_price"], aft=x["after_price"]); continue
            ev[d] = dict(m=1 + st, c=ca, note=f"現金股利{ca}元＋配股{st:.4g}股／股（FinMind TaiwanStockDividend；參考價驗算）")
        for c in CA.get(sid, []):
            d = c["恢復買賣日"]; ev[d] = dict(m=1 - float(c["減資比率"]), c=float(c["每股退還現金"]), note=f"{c['類型']}：每原股換{1-float(c['減資比率']):.4g}股、退還現金{c['每股退還現金']}元（MOPS）")
        CACHE[sid] = (px, ev)
    return CACHE[sid]

def valid(x): return x is not None and x["Trading_Volume"] > 0 and x["open"] > 0 and x["close"] > 0
def prev_valid(px, d):   # 回傳 (前一有效收盤, 相隔市場交易日數)
    i = CPOS[d] - 1
    while i >= 0:
        x = px.get(CAL[i])
        if valid(x): return x["close"], CPOS[d] - i
        i -= 1
    return None, 0
def prev_close(px, d):
    i = CPOS[d] - 1
    while i >= 0:
        x = px.get(CAL[i])
        if valid(x): return x["close"]
        i -= 1
    return None
def locked(px, d):   # 一字價且達漲跌停：日線無法證明成交
    x = px[d]; pc = prev_close(px, d)
    return x["open"] == x["max"] == x["min"] == x["close"] and pc is not None and abs(x["close"] / pc - 1) >= limit(d) - 0.005
def build(evs, tag):
    out, notes = [], []
    for r in evs:
        sid = r["公司代號"]; px, dv = stock(sid); ad = iso(r["發言日期"])
        end = PER.get(sid, {}).get("終止上市櫃日", ""); end = iso(end) if end else ""
        ei = next((i for i, d in enumerate(CAL) if d > ad), None)
        base = {"r": r, "tag": tag, "E": CAL[ei] if ei is not None else "", "H": {}}
        if ei is None: notes.append((r, "全部", "資料終點前無公告後交易日")); continue
        Ed = CAL[ei]; en = px.get(Ed)
        if not valid(en):
            why = "進場日無有效開盤（%s）" % ("無資料" if en is None else f"成交量{en['Trading_Volume']}、開盤{en['open']}")
            notes.append((r, "全部", why + "；依策略卡第11節不延後進場")); base["entry_fail"] = why; out.append(base); continue
        base["open"] = en["open"]; base["entry_lock"] = locked(px, Ed)
        for H in HS:
            xi = ei + H - 1
            if xi >= len(CAL): base["H"][H] = {"status": "未完成（尚未到出場日）", "sched": ""}; continue
            sched = CAL[xi]; j = xi; act = None; status = "正常"
            while j < len(CAL):
                d = CAL[j]
                if end and d >= end: break
                if valid(px.get(d)): act = d; break
                j += 1
            if act is None:
                if end and sched >= end or (end and j < len(CAL) and CAL[j] >= end):
                    last = next((CAL[k] for k in range(min(j, len(CAL)) - 1, ei, -1) if valid(px.get(CAL[k]))), None)
                    if last: act, status = last, f"下市（{end}）前以最終交易日收盤結算"
                    else: base["H"][H] = {"status": f"未解決：{end} 終止上市櫃，持有期間無最終交易價", "sched": sched}; continue
                else:
                    base["H"][H] = {"status": "未完成（順延後仍無有效收盤，資料終點）", "sched": sched}; continue
            elif act != sched:
                status = f"出場順延（預定 {sched} 無有效收盤）"
            # 跳價檢查（不含進場日；除權息日不檢查）
            jump = None
            k0 = CPOS[Ed]
            for k in range(k0 + 1, CPOS[act] + 1):
                d = CAL[k]; x = px.get(d)
                if not valid(x) or d in dv: continue
                pc, n = prev_valid(px, d)   # 中間無成交日時，容許 n 日累積漲跌停幅度
                up = (1 + limit(d)) ** n - 1 + 0.015; dn = 1 - (1 - limit(d)) ** n + 0.015
                if pc and any(v / pc - 1 > up or 1 - v / pc > dn for v in (x["open"], x["close"])):
                    jump = (d, pc, x["open"], x["close"]); break
            if jump:
                base["H"][H] = {"status": f"未解決：{jump[0]} 跳價（前收 {jump[1]} → 開 {jump[2]}／收 {jump[3]}）疑似公司行動未核對", "sched": sched}; continue
            sh, cash, notes_ca, bad = 1.0, 0.0, [], None
            for k in range(k0 + 1, CPOS[act] + 1):
                e = dv.get(CAL[k])
                if not e: continue
                if "bad" in e: bad = (CAL[k], e); break
                cash += sh * e["c"]; sh *= e["m"]; notes_ca.append(f"{CAL[k]} {e['note']}")
            ex = px[act]
            idx = IDX["TPEx" if r["市場別"] == "otc" else "TAIEX"]; ia, ib = idx.get(Ed), idx.get(act)
            mk = (ib["close"] / ia["open"] - 1) if ia and ib else None
            if bad:
                d_, e = bad; pr = ex["close"] / en["open"] - 1; lo = f"價格報酬{pr:+.4%}"; hi = None
                if "val" in e: hi = (ex["close"] + e["val"]) / en["open"] - 1; lo += f"；若除權價差（FinMind 權值{e['val']}元/原股）計為應收則約{hi:+.4%}（僅供界線，未納入）"
                base["H"][H] = {"status": f"未解決：{d_} {e['bad']}。{lo}", "sched": sched,
                                "alt": {"lo": pr, "hi": hi, "hold": CPOS[act] - k0 + 1, "mk": mk, "exit_lock": locked(px, act), "exit": act}}; continue
            g = (ex["close"] * sh + cash) / en["open"] - 1
            base["H"][H] = {"status": status, "sched": sched, "exit": act, "close": ex["close"], "sh": sh, "cash": cash, "ca": "；".join(notes_ca), "ret": g, "mk": mk,
                            "hold": CPOS[act] - k0 + 1, "exit_lock": locked(px, act)}
        out.append(base)
    return out, notes

def done(t, H): h = t["H"].get(H); return h is not None and "ret" in h
def holm(ps):
    idx = sorted(range(len(ps)), key=lambda i: ps[i] if ps[i] == ps[i] else 9); adj = [float("nan")] * len(ps); run = 0; k = len([p for p in ps if p == p])
    for rank, i in enumerate(idx):
        if ps[i] != ps[i]: continue
        run = max(run, min(1, (k - rank) * ps[i])); adj[i] = run
    return adj
def table(ts, label, kind, H, extra=None):
    xs, hd, ds = [], [], []
    for t in ts:
        if not done(t, H): continue
        h = t["H"][H]
        if extra and not extra(t, h): continue
        x = {"做多": h["ret"], "放空(做多取負)": -h["ret"], "做多超額(減大盤)": (h["ret"] - h["mk"]) if h["mk"] is not None else None}[kind]
        if x is None: continue
        xs.append(x); hd.append(h["hold"]); ds.append(t["E"])
    if len(xs) < 2: return None
    s = dict(summarize(xs, hold_days=hd, cluster=ds, cluster_unit="進場日"))
    s.update(組=label, 口徑=kind, 持有H=H, 中位數=median(xs), 最小值=min(xs), 最大值=max(xs),
             第5百分位=sorted(xs)[max(0, int(len(xs) * 0.05) - 1)] if len(xs) >= 20 else None)
    return s
